# Third-party dependencies for magpie.
#
# Produces two things the engine's own CMakeLists.txt links against:
#
#   magpie_ext    static library with every third-party implementation, plus the
#                 include paths the engine expects (<volk/volk.h>, <vma/vk_mem_alloc.h>,
#                 <SPIRV-Reflect/spirv_reflect.h>, <stb/...>, <gltf/cgltf.h>,
#                 <miniaudio/miniaudio.h>, <lua.h>, <imgui.h>, <SDL3/SDL.h>, ...)
#   magpie_slang  imported Slang compiler library + <slang/slang.h> include path
#
# Where things come from:
#   ext/include/, ext/lua/   copied from the C magpie repo's src/ext/ (cgltf, stb, miniaudio, lua 5.5)
#   FetchContent (git)       Vulkan-Headers, volk, SPIRV-Reflect, VMA, SDL3, Dear ImGui
#   Slang                    prebuilt release archive, unless SLANG_ROOT points at an install
#
# To use local copies instead of downloading (e.g. from the Vulkan SDK), set
# FETCHCONTENT_SOURCE_DIR_<NAME> (e.g. -DFETCHCONTENT_SOURCE_DIR_SDL3=C:/src/SDL), and
# SLANG_ROOT to a folder containing include/ and lib/ (both the SDK layout
# include/slang/slang.h and the release layout include/slang.h work).

include(FetchContent)

set(MAGPIE_VULKAN_SDK_TAG "vulkan-sdk-1.4.350.0" CACHE STRING "Tag used for Vulkan-Headers, volk and SPIRV-Reflect")
set(MAGPIE_SLANG_VERSION  "2026.22.1"            CACHE STRING "Slang release downloaded when SLANG_ROOT is not set")
set(SLANG_ROOT            ""                     CACHE PATH   "Existing Slang install (leave empty to download a release)")

# SOURCE_SUBDIR points at a folder that doesn't exist, so MakeAvailable only
# downloads; we compile the files we need ourselves below.
macro(magpie_fetch_sources name repo tag)
	FetchContent_Declare(${name} GIT_REPOSITORY ${repo} GIT_TAG ${tag} GIT_SHALLOW TRUE SOURCE_SUBDIR _no_cmake_)
	FetchContent_MakeAvailable(${name})
endmacro()

magpie_fetch_sources(vulkan_headers https://github.com/KhronosGroup/Vulkan-Headers.git  ${MAGPIE_VULKAN_SDK_TAG})
magpie_fetch_sources(volk           https://github.com/zeux/volk.git                    ${MAGPIE_VULKAN_SDK_TAG})
magpie_fetch_sources(spirv_reflect  https://github.com/KhronosGroup/SPIRV-Reflect.git   ${MAGPIE_VULKAN_SDK_TAG})
magpie_fetch_sources(vma            https://github.com/GPUOpen-LibrariesAndSDKs/VulkanMemoryAllocator.git v3.5.0)
magpie_fetch_sources(imgui          https://github.com/ocornut/imgui.git                v1.92.9b)

# SDL3: use the installed one if there is one (distro package, or anything on
# CMAKE_PREFIX_PATH). That already has the audio/X11/Wayland backends your
# system supports. Otherwise build it from source as a static library, which
# needs those backends' development packages (see the README).
option(MAGPIE_FETCH_SDL3 "Build SDL3 from source even if an installed SDL3 is found" OFF)

if (NOT MAGPIE_FETCH_SDL3)
	find_package(SDL3 3.2 CONFIG QUIET)
endif()

if (SDL3_FOUND)
	message(STATUS "magpie: using installed SDL3 ${SDL3_VERSION} (${SDL3_DIR})")
else()
	message(STATUS "magpie: no installed SDL3 found, building it from source")

	set(SDL_SHARED OFF CACHE BOOL "" FORCE)
	set(SDL_STATIC ON CACHE BOOL "" FORCE)
	set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
	set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
	FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.4.20 GIT_SHALLOW TRUE)
	FetchContent_MakeAvailable(SDL3)
endif()


# ------------------------------------------------------------------
# Include shims
#
# The engine includes some headers through a folder name (<volk/volk.h>,
# <vma/vk_mem_alloc.h>, ...) that the upstream repositories don't have.
# Each shim is a one-line header forwarding to the real file.
# ------------------------------------------------------------------

set(MAGPIE_SHIM_DIR ${CMAKE_CURRENT_BINARY_DIR}/shim_include)

function(magpie_shim relative_path real_path)
	set(out ${MAGPIE_SHIM_DIR}/${relative_path})
	set(text "// generated by ext/CMakeLists.txt\n#include \"${real_path}\"\n")
	if (EXISTS ${out})
		file(READ ${out} old)
	endif()
	if (NOT "${old}" STREQUAL "${text}")
		file(WRITE ${out} "${text}")
	endif()
endfunction()

magpie_shim(volk/volk.h                   ${volk_SOURCE_DIR}/volk.h)
magpie_shim(vma/vk_mem_alloc.h            ${vma_SOURCE_DIR}/include/vk_mem_alloc.h)
magpie_shim(SPIRV-Reflect/spirv_reflect.h ${spirv_reflect_SOURCE_DIR}/spirv_reflect.h)


# ------------------------------------------------------------------
# magpie_ext
# ------------------------------------------------------------------

file(GLOB MAGPIE_LUA_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/lua/*.c)

add_library(magpie_ext STATIC
	impl/ext_impl.cpp
	impl/vk_mem_alloc_impl.cpp
	impl/miniaudio_impl.c
	${volk_SOURCE_DIR}/volk.c
	${spirv_reflect_SOURCE_DIR}/spirv_reflect.c
	${MAGPIE_LUA_SOURCES}
	${imgui_SOURCE_DIR}/imgui.cpp
	${imgui_SOURCE_DIR}/imgui_draw.cpp
	${imgui_SOURCE_DIR}/imgui_tables.cpp
	${imgui_SOURCE_DIR}/imgui_widgets.cpp
	${imgui_SOURCE_DIR}/imgui_demo.cpp
	${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
	${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp
)

target_include_directories(magpie_ext PUBLIC
	${MAGPIE_SHIM_DIR}
	${CMAKE_CURRENT_SOURCE_DIR}/include
	${CMAKE_CURRENT_SOURCE_DIR}/lua
	${vulkan_headers_SOURCE_DIR}/include
	${volk_SOURCE_DIR}
	${imgui_SOURCE_DIR}
)

# Every Vulkan call goes through volk's function pointers.
target_compile_definitions(magpie_ext PUBLIC VK_NO_PROTOTYPES)
target_compile_definitions(magpie_ext PRIVATE IMGUI_IMPL_VULKAN_USE_VOLK)

if (WIN32)
	target_compile_definitions(magpie_ext PUBLIC VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()

target_link_libraries(magpie_ext PUBLIC SDL3::SDL3 ${CMAKE_DL_LIBS})

if (UNIX AND NOT APPLE)
	find_package(Threads REQUIRED)
	target_link_libraries(magpie_ext PUBLIC Threads::Threads m)
	target_compile_definitions(magpie_ext PRIVATE LUA_USE_LINUX)
elseif (APPLE)
	target_compile_definitions(magpie_ext PRIVATE LUA_USE_MACOSX)
endif()

# Third-party code: don't drown the build log in its warnings.
if (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
	target_compile_options(magpie_ext PRIVATE -w)
elseif (MSVC)
	target_compile_options(magpie_ext PRIVATE /W0)
endif()


# ------------------------------------------------------------------
# magpie_slang
# ------------------------------------------------------------------

if (NOT SLANG_ROOT)
	if (WIN32)
		set(_slang_platform "windows-x86_64.zip")
	elseif (APPLE)
		if (CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
			set(_slang_platform "macos-aarch64.zip")
		else()
			set(_slang_platform "macos-x86_64.zip")
		endif()
	else()
		if (CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
			set(_slang_platform "linux-aarch64.tar.gz")
		else()
			set(_slang_platform "linux-x86_64.tar.gz")
		endif()
	endif()

	FetchContent_Declare(slang_release
		URL https://github.com/shader-slang/slang/releases/download/v${MAGPIE_SLANG_VERSION}/slang-${MAGPIE_SLANG_VERSION}-${_slang_platform}
		SOURCE_SUBDIR _no_cmake_
		DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
	FetchContent_MakeAvailable(slang_release)

	set(_slang_root ${slang_release_SOURCE_DIR})
else()
	set(_slang_root ${SLANG_ROOT})
endif()

find_library(MAGPIE_SLANG_LIBRARY NAMES slang-compiler slang PATHS ${_slang_root}/lib NO_DEFAULT_PATH NO_CACHE)

if (NOT MAGPIE_SLANG_LIBRARY)
	message(FATAL_ERROR "Couldn't find the Slang library in ${_slang_root}/lib")
endif()

# The engine includes <slang/slang.h>; the release archives put slang.h straight in include/.
if (EXISTS ${_slang_root}/include/slang/slang.h)
	set(_slang_include ${_slang_root}/include)
else()
	magpie_shim(slang/slang.h         ${_slang_root}/include/slang.h)
	magpie_shim(slang/slang-com-ptr.h ${_slang_root}/include/slang-com-ptr.h)
	set(_slang_include ${MAGPIE_SHIM_DIR})
endif()

add_library(magpie_slang INTERFACE)
target_link_libraries(magpie_slang INTERFACE ${MAGPIE_SLANG_LIBRARY})
target_include_directories(magpie_slang INTERFACE ${_slang_include})

# Slang's shared libraries (and the ones it loads at runtime) have to sit next
# to the executable on Windows. Executables call magpie_copy_runtime(<target>).
if (WIN32)
	set(MAGPIE_SLANG_BIN_DIR ${_slang_root}/bin CACHE INTERNAL "")
endif()

function(magpie_copy_runtime target)
	if (WIN32 AND MAGPIE_SLANG_BIN_DIR)
		add_custom_command(TARGET ${target} POST_BUILD
			COMMAND ${CMAKE_COMMAND} -E copy_directory ${MAGPIE_SLANG_BIN_DIR} $<TARGET_FILE_DIR:${target}>
			COMMENT "Copying Slang runtime next to ${target}")
	endif()
endfunction()
